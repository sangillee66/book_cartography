# 함수
indicatrix_list_to_sf <- function(
    ii,
    crs_out,
    scale = 6e5,
    n = 120
) {
  
  ti_ellipse_fun <- get("ti_ellipse", envir = asNamespace("tissot"))
  indicatrix_to_sf_polygon <- function(z) {
    mat <- ti_ellipse_fun(z, scale = scale, n = n)
    mat <- as.matrix(mat)
    mat <- mat[complete.cases(mat), , drop = FALSE]
    if (nrow(mat) < 4) {
      return(st_polygon())
    }
    if (!all(mat[1, ] == mat[nrow(mat), ])) {
      mat <- rbind(mat, mat[1, ])
    }
    st_polygon(list(mat))
  }
  get_num <- function(z, name) {
    x <- z[[name]]
    if (is.null(x) || length(x) == 0) return(NA_real_)
    as.numeric(x[1])
  }
  st_sf(
    lon = map_dbl(ii, \(z) get_num(z, "lon")),
    lat = map_dbl(ii, \(z) get_num(z, "lat")),
    scale_h = map_dbl(ii, \(z) get_num(z, "scale_h")),
    scale_k = map_dbl(ii, \(z) get_num(z, "scale_k")),
    scale_a = map_dbl(ii, \(z) get_num(z, "scale_a")),
    scale_b = map_dbl(ii, \(z) get_num(z, "scale_b")),
    scale_area = map_dbl(ii, \(z) get_num(z, "scale_area")),
    angle_deformation = map_dbl(ii, \(z) get_num(z, "angle_deformation")),
    convergence = map_dbl(ii, \(z) get_num(z, "convergence")),
    geometry = st_sfc(
      map(ii, indicatrix_to_sf_polygon),
      crs = crs_out
    )
  )
}

# 경위도 좌표열을 투영 좌표열로 변환
project_xy <- function(xy) {
  st_sfc(
    st_linestring(xy),
    crs = 4326
  ) |>
    st_transform(crs_igh) |>
    st_coordinates() |>
    as.matrix() |>
    _[, 1:2]
}

# 투영된 좌표로 lobe polygon 만들기
make_igh_lobe_projected <- function(xmin, xmax, ymin, ymax, n = 720) {
  
  bottom_ll <- cbind(seq(xmin, xmax, length.out = n), rep(ymin, n))
  right_ll  <- cbind(rep(xmax, n), seq(ymin, ymax, length.out = n))
  top_ll    <- cbind(seq(xmax, xmin, length.out = n), rep(ymax, n))
  left_ll   <- cbind(rep(xmin, n), seq(ymax, ymin, length.out = n))
  
  bottom <- project_xy(bottom_ll)
  right  <- project_xy(right_ll)
  top    <- project_xy(top_ll)
  left   <- project_xy(left_ll)
  
  coords <- rbind(
    bottom,
    right[-1, ],
    top[-1, ],
    left[-1, ],
    bottom[1, ]
  )
  
  st_polygon(list(coords))
}

# ------------------------------------------------------------
# 2. 직접 투영 함수
# ------------------------------------------------------------

wrap_dlon <- function(lon, lon0) {
  ((lon - lon0 + 180) %% 360) - 180
}

moll_theta <- function(phi) {
  
  theta <- phi
  
  near_north <- abs(phi - pi / 2) < 1e-12
  near_south <- abs(phi + pi / 2) < 1e-12
  normal <- !(near_north | near_south)
  
  theta[near_north] <-  pi / 2
  theta[near_south] <- -pi / 2
  
  if (any(normal)) {
    th <- theta[normal]
    ph <- phi[normal]
    
    for (i in seq_len(12)) {
      th <- th -
        (2 * th + sin(2 * th) - pi * sin(ph)) /
        (2 + 2 * cos(2 * th))
    }
    
    theta[normal] <- th
  }
  
  theta
}

theta0 <- moll_theta(phi0_rad)
y0_sinu <- R * phi0_rad
y0_moll <- sqrt(2) * R * sin(theta0)
y_offset <- y0_sinu - y0_moll

goode_xy <- function(lon, lat, lon0, xoff = 0) {
  
  lat <- pmax(pmin(lat, 90 - 1e-10), -90 + 1e-10)
  
  lambda <- wrap_dlon(lon, lon0) * pi / 180
  phi <- lat * pi / 180
  
  x <- numeric(length(phi))
  y <- numeric(length(phi))
  
  low <- abs(lat) <= phi0
  high <- !low
  
  x[low] <- R * lambda[low] * cos(phi[low])
  y[low] <- R * phi[low]
  
  if (any(high)) {
    theta <- moll_theta(phi[high])
    x[high] <- (2 * sqrt(2) * R / pi) * lambda[high] * cos(theta)
    y_moll <- sqrt(2) * R * sin(theta)
    y[high] <- y_moll + sign(phi[high]) * y_offset
  }
  
  cbind(x + xoff, y)
}

project_mat <- function(mat, lon0, xoff) {
  goode_xy(mat[, 1], mat[, 2], lon0 = lon0, xoff = xoff)
}

project_sfg <- function(g, lon0, xoff) {
  
  gt <- as.character(st_geometry_type(g))
  
  if (st_is_empty(g)) {
    return(st_geometrycollection())
  }
  
  if (gt == "POINT") {
    st_point(as.numeric(project_mat(matrix(g, ncol = 2), lon0, xoff)))
    
  } else if (gt == "MULTIPOINT") {
    st_multipoint(project_mat(g, lon0, xoff))
    
  } else if (gt == "LINESTRING") {
    st_linestring(project_mat(g, lon0, xoff))
    
  } else if (gt == "MULTILINESTRING") {
    st_multilinestring(lapply(g, project_mat, lon0 = lon0, xoff = xoff))
    
  } else if (gt == "POLYGON") {
    st_polygon(lapply(g, project_mat, lon0 = lon0, xoff = xoff))
    
  } else if (gt == "MULTIPOLYGON") {
    st_multipolygon(
      lapply(g, function(poly) {
        lapply(poly, project_mat, lon0 = lon0, xoff = xoff)
      })
    )
    
  } else if (gt == "GEOMETRYCOLLECTION") {
    st_geometrycollection(
      lapply(g, project_sfg, lon0 = lon0, xoff = xoff)
    )
    
  } else {
    stop("지원하지 않는 geometry type입니다: ", gt)
  }
}

project_sf <- function(x, lon0, xoff) {
  
  if (nrow(x) == 0) {
    return(x)
  }
  
  geom2 <- st_sfc(
    lapply(st_geometry(x), project_sfg, lon0 = lon0, xoff = xoff)
  )
  
  st_sf(st_drop_geometry(x), geometry = geom2)
}

make_box_ll <- function(xmin, xmax, ymin, ymax, eps = 1e-5, n = 720) {
  
  bottom <- cbind(seq(xmin + eps, xmax - eps, length.out = n), ymin + eps)
  right  <- cbind(xmax - eps, seq(ymin + eps, ymax - eps, length.out = n))
  top    <- cbind(seq(xmax - eps, xmin + eps, length.out = n), ymax - eps)
  left   <- cbind(xmin + eps, seq(ymax - eps, ymin + eps, length.out = n))
  
  st_polygon(list(rbind(
    bottom,
    right[-1, ],
    top[-1, ],
    left[-1, ],
    bottom[1, ]
  )))
}

make_lobe_ll_parts <- function(xmin, xmax, ymin, ymax, eps = 1e-5, n = 720) {
  
  if (xmin < xmax) {
    st_sfc(
      make_box_ll(xmin, xmax, ymin, ymax, eps, n),
      crs = 4326
    )
  } else {
    st_sfc(
      make_box_ll(xmin, 180, ymin, ymax, eps, n),
      make_box_ll(-180, xmax, ymin, ymax, eps, n),
      crs = 4326
    )
  }
}

make_lobe_projected <- function(xmin, xmax, ymin, ymax, lon0, xoff, eps = 1e-5, n = 720) {
  
  if (xmin < xmax) {
    lon_left <- xmin + eps
    lon_right <- xmax - eps
  } else {
    lon_left <- xmin + eps
    lon_right <- xmax + 360 - eps
  }
  
  bottom <- cbind(seq(lon_left, lon_right, length.out = n), ymin + eps)
  right  <- cbind(lon_right, seq(ymin + eps, ymax - eps, length.out = n))
  top    <- cbind(seq(lon_right, lon_left, length.out = n), ymax - eps)
  left   <- cbind(lon_left, seq(ymax - eps, ymin + eps, length.out = n))
  
  ll <- rbind(
    bottom,
    right[-1, ],
    top[-1, ],
    left[-1, ],
    bottom[1, ]
  )
  
  ll[, 1] <- ((ll[, 1] + 180) %% 360) - 180
  ll[ll[, 1] == -180, 1] <- 180
  
  st_polygon(list(
    goode_xy(ll[, 1], ll[, 2], lon0 = lon0, xoff = xoff)
  ))
}
