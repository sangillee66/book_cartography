14.2.2 지도 디자인: 예시

14.2.2.1 데이터 준비

  • 형상 데이터 불러오기: 읍면동 변동을 반영한 데이터
emd_adj <- read_sf("D:/My R/Korean Administrative Areas/행정구역 셰이프 파일/3 Generalization/2020_Census_Adj/EMD_2020_Census_GEN_0020.shp", options = "ENCODING=CP949")
  • 속성 데이터 불러오기: 읍면동 변동을 반영한 데이터
pop_adj <- read_excel("D:/My R/Korea Census Data Manipulation/1 Population Basic/DM_ADJ_Census_2020.xlsx", sheet = 1)
  • 결합
emd_adj <- emd_adj |> 
  mutate(EMD_ID = as.numeric(EMD_ID))
emd_adj_data <- emd_adj |> 
  left_join(pop_adj, join_by(EMD_ID == Code)) |> 
  mutate(
    pop_t = ifelse(is.na(pop_t), 0, pop_t)
  )
  • 인구밀도 계산
emd_adj_data <- emd_adj_data |>
  mutate(
     area = geometry |> st_area() |> as.numeric()/1000000,
     pop_den = pop_t / area
  )
  • tmap: 지도 제작
bbox_ulleung_polygon <- bbox_ulleung |> 
  st_cast("POLYGON")

my_map <- tm_graticules(col = "white", x = seq(126, 130, 1), y = seq(33, 39, 1), labels.size = 0.75) +
  tm_shape(bbox_ulleung_polygon) + tm_fill(fill = "gray90") +
  tm_shape(emd_adj_data, is.main = TRUE) +
  tm_polygons(
    fill = "pop_den", col = NULL,
    fill.scale = tm_scale(
      breaks = c(0, 50, 250, 500, 1000, 5000, 10000, Inf),
      labels = c("0 - 50", "50 - 250", "250 - 500", "500 - 1,000", 
                 "1,000 - 5,000", "5,000 - 10,000", "10,000+"),
      values = "brewer.yl_gn_bu"
    ),
    fill.legend = tm_legend(
      title = expression("People/km"^2), 
      position = tm_pos_in(0.75, 0.28),
      title.size = 1.2, text.size = 1
    )
  ) +
  tm_shape(sigungu1) + tm_borders() +
  tm_shape(sido_polyline) + tm_lines(col = "black", lwd = 1.75) +
  tm_shape(bbox_seohae) + tm_borders() +
  tm_shape(bbox_ulleung_polygon) + tm_borders() +
  tm_title_in("Population Distribution, 2020", size = 2) +
  tm_layout(bg.color = "gray90", inner.margins = c(0.05, 0.09, 0.05, 0.05)) +
  tm_scalebar(breaks = seq(0, 200, 50), text.size = 0.6, color.dark = "gray60", 
             position = c(0.52, 0.03), allow_clipping = TRUE) +
  tm_credits("SANG-IL LEE, Geography Education at SNU", size = 0.7, position = c(0.025, 0.02))
